# バルカン戦争イベントチェーン
#
# 独露の代理戦争。HUN(独側) vs YUG(露側) + ROM参戦 → 少数民族蜂起 → 講和。
#
# イベントフロー:
#   .1(HUN,date) →45d→ .2(YUG,宣戦) →→→ .10(hidden,15d毎poll,sp>0.3)
#                                    |        → .11(蜂起, BNT/SLO/RUT独立)
#                          +3d→ .4(GER支援)    → .22(hidden,10d毎poll,sp>0.7|Budapest)
#                          +3d→ .5(RUS支援)        → .15(講和)
#                          +45d→ .8(ROM参戦)           +30d→ .18(バナト分割)
#                                                      +45d→ .20(ブルゲンラント割譲)
#
# ポーリング: 2段階制
#   .10 = 蜂起判定(sp>0.3) → 15日毎ループ
#   .22 = 講和判定(sp>0.7 or Budapest陥落) → 10日毎ループ
#   蜂起→講和の間に時間差を設けるため分離。
#
# ニュースイベント: .100(開戦), .101(ROM参戦), .102(終戦), .103(バナト分割), .107(ブルゲンラント割譲)
#####################################################################

### 第1幕：国境事変 ###
# date trigger + MTTH。1936.10〜12月にHUNで自然発火。
# ここでバルカン協商を結成し、ROMを事前加盟させる（開戦前に陣営が必要）。

country_event = {
	id = tsr_balkan_war.1
	title = tsr_balkan_war.1.t
	desc = tsr_balkan_war.1.d
	picture = GFX_report_event_PLACEHOLDER

	is_triggered_only = no

	trigger = {
		date > 1936.10.1
		NOT = { date > 1936.12.31 }
		tag = HUN
		NOT = { has_country_flag = tsr_balkan_war_started }
	}

	mean_time_to_happen = {
		days = 14
	}

	immediate = {
		set_country_flag = tsr_balkan_war_started
	}

	option = {
		name = tsr_balkan_war.1.a
		add_political_power = -25
		add_war_support = 0.05
		# ROMを開戦前にバルカン協商へ加盟させる（.8のadd_to_warの前提）。
		# 同時にAI参戦欲求抑制の国民精神を付与し、YUG開戦時にROMが
		# 巻き込まれて即時参戦するのを防ぐ。.8のadd_to_warで強制参戦時に自動cancel。
		ROM = {
			add_ideas = tsr_rom_opportunistic_neutrality
			add_equipment_to_stockpile = {
				type = infantry_equipment
				amount = 10000
				producer = RUS
			}
		}
		# 陣営結成: add_to_warにtargeted_allianceを使うため、開戦前に必要
		YUG = {
			add_equipment_to_stockpile = {
				type = infantry_equipment
				amount = 10000
				producer = RUS
			}
			create_faction_from_template = faction_template_balkan_entente
			add_to_faction = ROM
			country_event = { id = tsr_balkan_war.2 days = 45 }
		}
		# HUN側: 洪勃防衛協定を結成しBULを加盟（防衛協定の事前締結）。
		# YUGがHUN(盟主)に宣戦すると、協定によりBULが自動的に防衛参戦する。
		create_faction_from_template = faction_template_hun_bul_defense_pact
		add_to_faction = BUL
		# 開戦前フレーバー: マジャル民族主義の高揚
		country_event = { id = tsr_balkan_war.30 days = 10 }
	}
}

### 第2幕：開戦 ###
# チェーンの起点。ここから全後続イベントの遅延発火を一括予約。
# declare_war_onでAI判断に依存せず確実に開戦させる。

country_event = {
	id = tsr_balkan_war.2
	title = tsr_balkan_war.2.t
	desc = tsr_balkan_war.2.d
	picture = GFX_report_event_PLACEHOLDER

	is_triggered_only = yes

	option = {
		name = tsr_balkan_war.2.a
		add_war_support = 0.1
		declare_war_on = {
			target = HUN
			type = annex_everything
		}
		HUN = { country_event = { id = tsr_balkan_war.3 days = 1 } }
		GER = { country_event = { id = tsr_balkan_war.4 days = 3 } }
		RUS = { country_event = { id = tsr_balkan_war.5 days = 3 } }
		ROM = { country_event = { id = tsr_balkan_war.8 days = 45 } }
		# 蜂起ポーリング開始: 開戦30日後から15日毎にsp>0.3をチェック
		HUN = { country_event = { id = tsr_balkan_war.10 days = 30 } }
		news_event = { id = tsr_balkan_war.100 days = 1 }
		# フレーバー: YUG開戦の大義、HUN少数民族地域の不穏（蜂起伏線）
		country_event = { id = tsr_balkan_war.31 days = 2 }
		HUN = { country_event = { id = tsr_balkan_war.37 days = 40 } }
		# 南部戦線: BULは洪勃防衛協定により自動防衛参戦。戦中精神を付与し通知。
		BUL = { add_ideas = tsr_bul_defensive_pact }
		BUL = { country_event = { id = tsr_balkan_war.50 days = 1 } }
		HUN = { country_event = { id = tsr_balkan_war.51 days = 1 } }
		news_event = { id = tsr_balkan_war.104 days = 2 }
		# GREは秘密協定により開戦75日後にBULの背後を突いて参戦
		GRE = { country_event = { id = tsr_balkan_war.52 days = 75 } }
	}
}

### 第5幕：内部崩壊 ###
# 2段階ポーリングの第1段階。
# .10: sp>0.3で蜂起 → .22: sp>0.7|Budapest陥落で講和
# immediate内で分岐し、条件未達なら自身を再予約（15日ループ）。

# 蜂起判定ポーリング（hidden, 15日毎）
country_event = {
	id = tsr_balkan_war.10
	title = tsr_balkan_war.10.t
	desc = tsr_balkan_war.10.d

	hidden = yes
	is_triggered_only = yes

	trigger = {
		has_war_with = YUG
		NOT = { has_country_flag = tsr_balkan_subversion_fired }
	}

	immediate = {
		if = {
			limit = { surrender_progress > 0.3 }
			set_country_flag = tsr_balkan_subversion_fired
			country_event = { id = tsr_balkan_war.11 days = 5 }
		}
		# 条件未達: 15日後に自身を再発火（ループ）
		if = {
			limit = {
				NOT = { surrender_progress > 0.3 }
				has_war_with = YUG
			}
			country_event = { id = tsr_balkan_war.10 days = 15 }
		}
	}

	option = {
		name = tsr_balkan_war.10.a
	}
}

# 講和判定ポーリング（hidden, 10日毎） — 第2段階
# province 9660 = ブダペスト。陥落 or sp>0.7 で講和発動。
# .10(蜂起)→.22(講和) と分離することで、蜂起と講和の間に最低10日の猶予を保証。
country_event = {
	id = tsr_balkan_war.22
	title = tsr_balkan_war.22.t
	desc = tsr_balkan_war.22.d

	hidden = yes
	is_triggered_only = yes

	trigger = {
		has_war_with = YUG
	}

	immediate = {
		if = {
			limit = {
				OR = {
					surrender_progress > 0.8
					NOT = { controls_province = 9660 }  # Budapest
				}
			}
			country_event = { id = tsr_balkan_war.15 days = 3 }
		}
		if = {
			limit = {
				NOT = { surrender_progress > 0.8 }
				controls_province = 9660
				has_war_with = YUG
			}
			country_event = { id = tsr_balkan_war.22 days = 10 }
		}
	}

	option = {
		name = tsr_balkan_war.22.a
	}
}
