		}

		visible = {
			has_country_flag = preparation_for_far_left_civil_war

				has_idea_with_trait = far_left_revolutionary
				has_idea_with_trait = ambitious_union_boss

			check_variable = { num_owned_states > 1 }
		}

		fire_only_once = yes

		cost = 50

		complete_effect = {
			if = {
				limit = {
					has_government = left
				}
				set_variable = {
					var = civil_war_size_var
					value = party_popularity@left
				}
			}
			else_if = {
				limit = {
					has_government = center_left
				}
				set_variable = {
					var = civil_war_size_var
					value = party_popularity@center_left
				}
			}
			else_if = {
				limit = {
					has_government = center
				}
				set_variable = {
					var = civil_war_size_var
					value = party_popularity@center
				}
			}
			else_if = {
				limit = {
					has_government = center_right
				}
				set_variable = {
					var = civil_war_size_var
					value = party_popularity@center_right
				}
			}
			else_if = {
				limit = {
					has_government = right
				}
				set_variable = {
					var = civil_war_size_var
					value = party_popularity@right
				}
			}
			else_if = {
				limit = {
					has_government = far_right
				}
				set_variable = {
					var = civil_war_size_var
					value = party_popularity@far_right
				}
			}
			subtract_from_variable = {
				var = civil_war_size_var
				value = army_support_var
			}
			if = {
				limit = {
					check_variable = {
						var = civil_war_size_var
						value = 0.3
						compare = less_than
					}
				}
				set_variable = {
					var = civil_war_size_var
					value = 0.3
				}
			}
			start_civil_war = {
				ruling_party = far_left
				ideology = ROOT
				size = civil_war_size_var
				keep_unit_leaders_trigger = {
					has_trait = hidden_sympathies
				}
			}
			clr_country_flag = preparation_for_far_left_civil_war
			clr_country_flag = military_support_for_far_left_civil_war
			clr_country_flag = civil_support_for_far_left_civil_war
			set_country_flag = ideology_civil_war
		}

		ai_will_do = {
			factor = 0
		}
	}
